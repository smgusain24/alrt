```typescript
"use client";

import { useState, useEffect, useCallback } from "react";
import {
    RetroButton, Badge, WindowCard, BeveledInput, RetroModal, GrooveDivider

import { Plus, Copy, Check, Trash2, AlertTriangle } from "lucide-react";

interface MemberRecord {
    id: string;
    email: string;
    name: string | null;
    role: string;
    created_at: string;
    last_login_at: string | null;
    record_type: string;
}

function relativeDate(dateStr: string): string {
    const date = new Date(dateStr);
    const now = new Date();
    const diffMs = now.getTime() - date.getTime();
    const diffMins = Math.floor(diffMs / 60000);
    if (diffMins < 1) return "Just now";
    if (diffMins < 60) return `${diffMins}m ago`;
    const diffHours = Math.floor(diffMins / 60);
    if (diffHours < 24) return `${diffHours}h ago`;
    const diffDays = Math.floor(diffHours / 24);
    return diffDays < 30 ? `${diffDays}d ago` : date.toLocaleDateString();
}

export default function MembersPage() {
    const [members, setMembers] = useState<MemberRecord[]>([]);
    const [teamId, setTeamId] = useState<string | null>(null);
    const [currentUserRole, setCurrentUserRole] = useState<string>("viewer");
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState<string | null>(null);

    // Invite modal state
    const [showInviteModal, setShowInviteModal] = useState(false);
    const [inviteEmail, setInviteEmail] = useState("");
    const [inviteRole, setInviteRole] = useState<"admin" | "viewer">("viewer");
    const [inviteUrl, setInviteUrl] = useState<string | null>(null);
    const [inviting, setInviting] = useState(false);
    const [inviteError, setInviteError] = useState<string | null>(null);
    const [copied, setCopied] = useState(false);

    // Delete invite state
    const [deletingInviteId, setDeletingInviteId] = useState<string | null>(null);

    const fetchMembers = useCallback(async (tid: string) => {
        const res: any = await api.invites.listMembers(tid);
        setMembers(res.members);
    }, []);

    useEffect(() => {
        api.auth.me()
            .then(async (user: any) => {
                setTeamId(user.team_id);
                setCurrentUserRole(user.role || "viewer");
                await fetchMembers(user.team_id);
            })
            .catch((err: Error) => setError(err.message))
            .finally(() => setLoading(false));
    }, [fetchMembers]);

    const openInviteModal = () => {
        setInviteEmail("");
        setInviteRole("viewer");
        setInviteUrl(null);
        setInviteError(null);
        setCopied(false);
        setShowInviteModal(true);
    };

    const closeInviteModal = () => {
        setShowInviteModal(false);
        if (inviteUrl && teamId) fetchMembers(teamId);
    };

    const handleInvite = async () => {
        if (!teamId || !inviteEmail.trim()) return;
        setInviting(true);
        setInviteError(null);
        try {
            const res: any = await api.invites.create(teamId, { email: inviteEmail.trim(), role: inviteRole });
            setInviteUrl(res.invite_url);
        } catch (err: any) {
            setInviteError(err.message);
        } finally {
            setInviting(false);
        }
    };

    const handleCopyLink = async () => {
        if (!inviteUrl) return;
        await navigator.clipboard.writeText(inviteUrl);
        setCopied(true);
        setTimeout(() => setCopied(false), 2000);
    };

    const handleDeleteInvite = async (inviteId: string) => {
        if (!teamId) return;
        setDeletingInviteId(inviteId);
        try {
            await api.invites.deleteInvite(teamId, inviteId);
            setMembers((prev) => prev.filter((m) => m.id !== inviteId));
        } catch (err: any) {
            setError(err.message);
        } finally {
            setDeletingInviteId(null);
        }
    };

    const isAdmin = currentUserRole === "admin";

    return (
        <div className="max-w-5xl mx-auto">
            <div className="flex items-center justify-between mb-6">
                <h1 className="font-heading text-2xl uppercase">Settings — Members</h1>
                {isAdmin && (
                    <RetroButton variant="accent" onClick={openInviteModal}>
                        <Plus className="w-4 h-4 inline mr-1" strokeWidth={2} />
                        Invite Member
                    </RetroButton>
                )}
            </div>

            <p className="text-foreground mb-4 font-mono text-sm">
                Team members share access to all workflows, subscribers, and notification data.
                Admins can invite and manage members. Viewers have read-only access.
            </p>

            <GrooveDivider />

            <div className="mt-4">
                {loading ? (
                    <p className="text-muted font-mono text-sm">Loading members...</p>
                ) : error ? (
                    <WindowCard title="Error">
                        <p className="text-danger font-mono text-sm">{error}</p>
                    </WindowCard>
                ) : (
                    <WindowCard title={`TEAM MEMBERS (${members.length})`}>
                        <div className="overflow-x-auto">
                            <table className="w-full text-sm">
                                <thead>
                                    <tr className="bg-[#c0c0c0] bevel-outset">
                                        <th className="px-3 py-2 text-left text-[10px] font-bold uppercase tracking-wider">Email</th>
                                        <th className="px-3 py-2 text-left text-[10px] font-bold uppercase tracking-wider">Name</th>
                                        <th className="px-3 py-2 text-left text-[10px] font-bold uppercase tracking-wider">Role</th>
                                        <th className="px-3 py-2 text-left text-[10px] font-bold uppercase tracking-wider">Status</th>
                                        <th className="px-3 py-2 text-left text-[10px] font-bold uppercase tracking-wider">Joined</th>
                                        {isAdmin && <th className="px-3 py-2 text-left text-[10px] font-bold uppercase tracking-wider">Actions</th>}
                                    </tr>
                                </thead>
                                <tbody>
                                    {members.map((m, idx) => (
                                        <tr key={m.id} className={`border-b border-muted/20 ${idx % 2 === 0 ? "bg-white" : "bg-row-alt"}`}>
                                            <td className="px-3 py-2 font-mono text-xs">{m.email}</td>
                                            <td className="px-3 py-2 font-mono text-xs">{m.name || <span className="text-muted">—</span>}</td>
                                            <td className="px-3 py-2">
                                                <Badge variant={m.role === "admin" ? "success" : "default"}>
                                                    {m.role}
                                                </Badge>
                                            </td>
                                            <td className="px-3 py-2">
                                                <Badge variant={m.record_type === "member" ? "success" : "warning"}>
                                                    {m.record_type === "member" ? "active" : "invited"}
                                                </Badge>
                                            </td>
                                            <td className="px-3 py-2 font-mono text-xs text-muted">
                                                {relativeDate(m.created_at)}
                                            </td>
                                            {isAdmin && (
                                                <td className="px-3 py-2">
                                                    {m.record_type === "invite" && (
                                                        <RetroButton
                                                            variant="danger"
                                                            className="!px-2 !py-1 !text-xs"
                                                            disabled={deletingInviteId === m.id}
                                                            onClick={() => handleDeleteInvite(m.id)}

            </div>
```

```typescript
"use client";

import { useEffect, useState } from "react";
import { useSearchParams, useRouter } from "next/navigation";
import { BeveledInput, RetroButton, WindowCard } from "@/components/retro";
import { api } from "@/lib/api";

export default function AcceptInvitePage() {
    const searchParams = useSearchParams();
    const router = useRouter();
    const token = searchParams.get("token") || "";

    const [inviteInfo, setInviteInfo] = useState<{ email: string; role: string } | null>(null);
    const [loadError, setLoadError] = useState<string | null>(null);
    const [password, setPassword] = useState("");
    const [name, setName] = useState("");
    const [submitting, setSubmitting] = useState(false);
    const [submitError, setSubmitError] = useState<string | null>(null);

    useEffect(() => {
        if (!token) {
            setLoadError("Invalid invite link — no token provided.");
            return;
        }
        api.invites.getInviteInfo(token)
            .then((res: any) => setInviteInfo({ email: res.email, role: res.role }))
            .catch((err: Error) => setLoadError(err.message || "Invalid or expired invite link."));
    }, [token]);

    const handleAccept = async (e: React.FormEvent) => {
        e.preventDefault();
        if (!token || !password) return;
        setSubmitting(true);
        setSubmitError(null);
        try {
            const res: any = await api.invites.acceptInvite({ token, password, name: name || undefined });
            if (res.token) {
                // Store token for immediate dashboard access
                localStorage.setItem("alrt_token", res.token);
            }
            router.push("/workflows");
        } catch (err: any) {
            setSubmitError(err.message || "Failed to accept invite. Please try again.");
        } finally {
            setSubmitting(false);
        }
    };

    return (
        <div className="min-h-screen bg-surface flex items-center justify-center p-4">
            <div className="w-full max-w-md">
                <div className="text-center mb-8">
                    <h1 className="font-heading text-3xl uppercase text-navy mb-1">ALRT.dev</h1>
                    <p className="font-mono text-xs text-muted uppercase tracking-widest">Team Invite</p>
                </div>

                <WindowCard title="ACCEPT INVITE">
                    {loadError ? (
                        <div className="text-center py-8 font-mono text-sm text-danger">{loadError}</div>
                    ) : !inviteInfo ? (
                        <div className="text-center py-8 font-mono text-xs text-muted animate-pulse uppercase tracking-widest">
                            Validating invite...
                        </div>
                    ) : (
                        <form onSubmit={handleAccept} className="flex flex-col gap-4 p-2">
                            <div className="bevel-inset bg-row-alt p-3 font-mono text-sm">
                                <div><span className="text-muted text-xs uppercase">Email:</span> {inviteInfo.email}</div>
                                <div><span className="text-muted text-xs uppercase">Role:</span> {inviteInfo.role}</div>
                            </div>

                            <BeveledInput
                                id="invite-name"
                                label="Your Name (optional)"
                                placeholder="Jane Smith"
                                value={name}
                                onChange={(e: React.ChangeEvent<HTMLInputElement>) => setName(e.target.value)}
                            />

                            <BeveledInput
                                id="invite-password"
                                label="Set Password"
                                type="password"
                                placeholder="Choose a strong password"
                                value={password}
                                onChange={(e: React.ChangeEvent<HTMLInputElement>) => setPassword(e.target.value)}
                                required
                            />

                            {submitError && (
                                <p className="text-danger font-mono text-xs">{submitError}</p>
                            )}

                            <RetroButton
                                type="submit"
                                variant="accent"
                                disabled={submitting || !password}
                                className="w-full"
                            >
                                {submitting ? "Setting up account..." : "Join Team"}
                            </RetroButton>
                        </form>
                    )}
                </WindowCard>
            </div>
        </div>
    );
}
```
